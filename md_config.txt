# =====================================================================
# ARCHIVO DE CONFIGURACIÓN DE DINÁMICA MOLECULAR (GROMACS)

# Este archivo permite especificar la lista de complejos que se simularán
# de forma secuencial, junto con sus parámetros físicos y globales.
#
# El script buscará automáticamente los archivos correspondientes en:
#   - PDB del Complejo: data/complexes/complex_{Nombre_Complejo}.pdb
#                       o data/complexes/{Nombre_Complejo}.pdb
#   - SDF del Ligando : data/ligands/{Ligando}_3D.sdf
#                       o data/ligands/{Ligando}.sdf
#                       (El ligando se extrae tras el carácter '_')
#
# =====================================================================
# 1. PARÁMETROS GLOBALES DE HARDWARE
# =====================================================================
# ntomp: Número de hilos OpenMP de CPU por tarea de mdrun (dejar en 8 para RTX 4080 Super)
ntomp = 8

# gpu_id: Identificador de la GPU NVIDIA a utilizar (usar 0 por defecto)
gpu_id = 0

# use_gpu: True para usar aceleración por GPU (CUDA), False para CPU
use_gpu = True

# update_mode: 'cpu' o 'gpu' (se recomienda 'cpu' por estabilidad en equilibración NPT)
update_mode = cpu

# =====================================================================
# 2. LISTA DE SIMULACIONES A EJECUTAR
# =====================================================================
# Formato:
#   Nombre_Complejo | Tiempo_ns | Compresibilidad_NPT | Maxwarn
#
# Parámetros:
#   - Nombre_Complejo: Identificador del complejo (ej. MurG_Afzelin, PBP2a_Ceftaroline).
#   - Tiempo_ns: Duración de la simulación de producción en nanosegundos (ej. 20 o 100).
#   - Compresibilidad_NPT: Compresibilidad isotrópica en bar^-1 para NPT (por defecto 4.5e-5).
#                          Nota: Para PBP2a_Ceftaroline se recomienda usar 4.5e-6 por estabilidad.
#   - Maxwarn: Número máximo de warnings permitidos en grompp (por defecto 1).
# ---------------------------------------------------------------------
